function doGet() { return HtmlService.createHtmlOutputFromFile('Index') .setTitle('نظام BAVLY لإدارة المصنع') .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL) .addMetaTag('viewport', 'width=device-width, initial-scale=1'); } // جلب كل البيانات دفعة واحدة عند فتح البرنامج function getInitialData() { var ss = SpreadsheetApp.getActiveSpreadsheet(); return { models: getSheetData(ss, "Models", ["كود الموديل", "اسم الموديل", "القسم", "الخامة", "التكلفة", "سعر البيع"]), customers: getSheetData(ss, "Customers", ["كود العميل", "اسم العميل", "الهاتف", "المحافظة", "إجمالي المشتريات", "المتبقي"]), materials: getSheetData(ss, "Materials", ["كود الخامة", "اسم الخامة", "الوحدة", "الكمية الحالية"]), invoices: getSheetData(ss, "SalesInvoices", ["رقم الفاتورة", "التاريخ", "اسم العميل", "الإجمالي", "الخصم", "الصافي", "طريقة الدفع"]) }; } function getSheetData(ss, sheetName, defaultHeaders) { var sheet = getOrCreateSheet(ss, sheetName, defaultHeaders); var data = sheet.getDataRange().getValues(); if (data.length <= 1) return []; var headers = data[0]; return data.slice(1).map(function(row) { var obj = {}; headers.forEach(function(h, i) { obj[h] = row[i]; }); return obj; }); } // حفظ الفاتورة بالكامل في صفحة المبيعات وصفحة العناصر function saveInvoice(invoiceHeader, invoiceItems) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var invSheet = getOrCreateSheet(ss, "SalesInvoices", ["رقم الفاتورة", "التاريخ", "اسم العميل", "الإجمالي", "الخصم", "الصافي", "طريقة الدفع"]); var itemsSheet = getOrCreateSheet(ss, "SalesItems", ["ID", "رقم الفاتورة", "الموديل", "اللون", "المقاس", "الكمية", "السعر", "الإجمالي"]); invSheet.appendRow(invoiceHeader); if (invoiceItems && invoiceItems.length > 0) { var rows = invoiceItems.map(function(item, idx) { return [ "ITEM-" + Date.now() + "-" + idx, invoiceHeader[0], item.model, item.color || "", item.size || "", Number(item.qty) || 1, Number(item.price) || 0, (Number(item.qty) || 1) * (Number(item.price) || 0) ]; }); itemsSheet.getRange(itemsSheet.getLastRow() + 1, 1, rows.length, rows[0].length).setValues(rows); } return { success: true, message: "تم حفظ الفاتورة بنجاح!" }; } // حفظ عنصر جديد (موديل / عميل / خامة) function saveSingleRecord(sheetName, headers, rowData) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = getOrCreateSheet(ss, sheetName, headers); sheet.appendRow(rowData); return { success: true, message: "تم الحفظ بنجاح!" }; } function getOrCreateSheet(ss, sheetName, headers) { var sheet = ss.getSheetByName(sheetName); if (!sheet) { sheet = ss.insertSheet(sheetName); sheet.appendRow(headers); sheet.getRange(1, 1, 1, headers.length).setBackground("#1e293b").setFontColor("#ffffff").setFontWeight("bold"); sheet.setFrozenRows(1); } return sheet; }
